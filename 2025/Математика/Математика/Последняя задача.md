$$
u=\ln\frac{1}{\sqrt{x^2+y^2}} = -\frac{1}{2}\ln(x^2+y^2)
$$
$$
\frac{\partial ^2 u}{\partial x^2}+\frac{\partial ^2 u}{\partial y^2}
$$
$$ \frac{\partial u}{\partial x} =  -\frac{1}{2}(\frac{1}{x^2+y^2})(x^2+y^2)'_x=-\frac{1}{2}(\frac{1}{x^2+y^2})2x=-\frac{x}{x^2+y^2}$$
$$\frac{\partial^2u}{\partial x^2}=-\frac{x^2+y^2-x\cdot2x}{(x^2+y^2)^2}=-\frac{y^2-x^2}{(x^2+y^2)^2}=\frac{x^2-y^2}{(x^2+y^2)^2}$$
$$ \frac{\partial u}{\partial y} =  -\frac{1}{2}(\frac{1}{x^2+y^2})(x^2+y^2)'_y=-\frac{1}{2}(\frac{1}{x^2+y^2})2y=-\frac{y}{x^2+y^2}$$
$$
\frac{\partial^2u}{\partial y^2}=-\frac{x^2+y^2-y\cdot2y}{(x^2+y^2)^2}=\frac{x^2-y^2}{(y^2+x^2)^2}
$$
$$
\frac{\partial ^2 u}{\partial x^2}+\frac{\partial ^2 u}{\partial y^2}=\frac{x^2-y^2+y^2-x^2}{(x^2+y^2)^2}=0
$$
